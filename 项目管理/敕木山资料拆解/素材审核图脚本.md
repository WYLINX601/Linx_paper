function wrap(text, width) {
  return String(text).split('\n').map(p => {
    let line = '', lines = [];
    for (const ch of p) {
      if (line && ea.measureText(line + ch).width > width) { lines.push(line); line = ch; }
      else line += ch;
    }
    lines.push(line); return lines.join('\n');
  }).join('\n');
}

  const id=ea.addText(x,y,wrap(value,w),{autoResize:true});
  const e=ea.getElement(id); if(link)e.link=link;
  return {id,h:e.height,w:e.width};
}
function linebox(x,y,w,h,color='#dbe6ed') {
