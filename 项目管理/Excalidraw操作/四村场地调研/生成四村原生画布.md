const textBoxes = [], panelBoxes = [], imageBoxes = [];
function text(x,y,value,width,size=24,color='#243447',link=null) {
  ea.style.fontFamily = 2; // installed EA reference: Helvetica + system CJK fallback
  ea.style.fontSize = size;
  ea.style.strokeColor = color;
  // Fixed-width addText can reflow later during export without updating the
  // earlier layout heights. Wrap with installed font metrics BEFORE placement.
  const lines=[];
  for(const line of value.split('\n')) {
    let current='';
    for(const char of Array.from(line)) {
      const candidate=current+char;
      if(current && ea.measureText(candidate).width > width-12) {
        lines.push(current);current=char;
      } else current=candidate;
    }
    lines.push(current);
  }
  const id = ea.addText(x,y,lines.join('\n'),{autoResize:true,textAlign:'left',textVerticalAlign:'top'});
  const el = ea.elementsDict[id];
  if (!el || !Number.isFinite(el.height) || el.height <= 0) throw new Error('Native text measurement failed');
  if(el.width>width+1) throw new Error('Measured text exceeds its assigned width');
  if (link) el.link=link;
  textBoxes.push({id,x,y,width,height:el.height});
  return {id,bottom:y+el.height,height:el.height};
}
function rect(x,y,w,h,fill,stroke='#cad4df',sw=1) {
  ea.style.backgroundColor=fill;ea.style.strokeColor=stroke;ea.style.fillStyle='solid';ea.style.strokeWidth=sw;ea.style.roughness=0;
  return ea.addRect(x,y,w,h);
}
async function panel(p,x,y,w) {
  const bg=rect(x,y,w,500,p.tint,p.accent,2);
  const stripe=rect(x,y,12,500,p.accent,p.accent,0);
  let cy=text(x+38,y+30,p.title,w-76,46,p.accent).bottom+18;
  cy=text(x+38,cy,p.scope,w-76,23,'#33465b').bottom+30;
  const gap=28,pad=32,cw=(w-pad*2-gap)/2;
  if (p.images.length) {
    let next=cy;
    for (let i=0;i<p.images.length;i++) {
      const image=p.images[i];
      if (!allowedAssets.has(image.path)) throw new Error('Unapproved image path');
      const f=app.vault.getAbstractFileByPath(image.path);
      if (!f) throw new Error('Missing image: '+image.path);
      const ix=x+pad+i*(cw+gap),iy=cy;
      rect(ix,iy,cw,310,'#ffffff','#d3dce4',1);
      const id=await ea.addImage({topX:ix+12,topY:iy+12,imageFile:f,scale:false,anchor:false});
      const e=ea.elementsDict[id];
      if (!e || !(e.width>0 && e.height>0)) throw new Error('Invalid native image');
      const scale=Math.min((cw-24)/e.width,286/e.height);
      e.width*=scale;e.height*=scale;e.x=ix+(cw-e.width)/2;e.y=iy+(310-e.height)/2;
      e.link='[['+image.path+']]';
      imageBoxes.push({id,path:image.path,x:e.x,y:e.y,width:e.width,height:e.height});
      const caption=text(ix+4,iy+324,image.caption,cw-8,19,'#526477','[['+image.path+']]');
      next=Math.max(next,caption.bottom+28);
    }
    cy=next;
  } else {
    const empty=rect(x+pad,cy,w-pad*2,130,'#ffffff','#d5dce4',1);
    text(x+pad+28,cy+24,'暂无可追溯的专属图片或案例档案\n不以其他村落的照片、栋数或地名填充',w-pad*2-56,28,'#586574');
    cy+=158;
  }
  for (let row=0;row<p.cards.length;row+=2) {
    const items=[];
    for (let j=0;j<2 && row+j<p.cards.length;j++) {
      const c=p.cards[row+j],cx=x+pad+j*(cw+gap),ids=[];
      const box=rect(cx,cy,cw,200,'#ffffff','#d2dce4',1);ids.push(box);
      let t=text(cx+22,cy+20,c.title,cw-44,28,p.accent);ids.push(t.id);
      t=text(cx+22,t.bottom+14,c.body,cw-44,24,'#243447');ids.push(t.id);
      t=text(cx+22,t.bottom+18,c.cite,cw-44,18,'#526b82',c.link);ids.push(t.id);
      items.push({box,ids,bottom:t.bottom+22});
    }
    const bottom=Math.max(...items.map(i=>i.bottom),cy+240);
    for(const i of items){ea.elementsDict[i.box].height=bottom-cy;ea.addToGroup(i.ids);}
    cy=bottom+22;
  }
  cy=text(x+pad,cy+6,'来源索引｜点击文字回查主源',w-pad*2,26,p.accent).bottom+12;
  for(const s of p.sources) cy=text(x+pad,cy,s.text,w-pad*2,18,'#41576c',s.link).bottom+12;
  if(p.key==='huishantou') {
    cy+=22;
    const h=rect(x+pad,cy,w-pad*2,270,'#ffffff','#c4ceda',1);
    const label=text(x+pad+24,cy+20,'原始笔记（原文保留；日期与天数待核）',w-pad*2-48,26,'#586574');
    if(job.originalElements.length!==1 || job.originalElements[0].id!=='uFs53dH0') throw new Error('Original scene changed; inspect');
    const original=structuredClone(job.originalElements[0]);
    original.x=x+pad+28;original.y=label.bottom+16;
    ea.elementsDict[original.id]=original;
    cy=Math.max(cy+270,original.y+original.height+26)+20;
  }
  const height=cy-y+24;
  ea.elementsDict[bg].height=height;ea.elementsDict[stripe].height=height;
  panelBoxes.push({key:p.key,x,y,width:w,height,bg,stripe});
  return height;
}
