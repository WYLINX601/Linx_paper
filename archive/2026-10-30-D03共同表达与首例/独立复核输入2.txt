def patterns(scheme):
    out=[]
    for rule in scheme['rules']:
        p=rule['pattern']; variants=[p]
        if scheme['allowReflection']: variants.append(list(reversed(p)))
        keys=set()
        for v in variants:
            for i in range(len(v)) if scheme['allowRotation'] else [0]: keys.add('|'.join(v[i:]+v[:i]))
        out.extend((k,rule['code']) for k in sorted(keys))
    return out

def classify(kinds,states,scheme,valid=True,shape='rectangular-reference',ids=_DEFAULT_IDS):
    if ids is _DEFAULT_IDS: ids=scheme['sideIds']
    arrays=[kinds,states,ids]
    if (valid is not True or not isinstance(shape,str) or not shape
        or any(not isinstance(v,list) or not v or any(not isinstance(x,str) or not x for x in v) for v in arrays)
        or not len(kinds)==len(states)==len(ids) or len(ids)!=len(set(ids))
        or any(k not in KINDS for k in kinds) or any(s not in ['known','unknown'] for s in states)
        or any((k=='unknown')!=(s=='unknown') for k,s in zip(kinds,states))): return 'invalid-input',[]
    if shape!=scheme['applicableShape'] or len(ids)!=len(scheme['sideIds']): return 'out-of-scope',[]
    if ids!=scheme['sideIds']: return 'invalid-input',[]
    if 'unknown' in states: return 'unknown',[]
    matches=sorted(set(code for key,code in patterns(scheme) if key=='|'.join(kinds)))
    return ('known-unmatched' if not matches else 'rule-conflict' if len(matches)>1 else 'candidate-assigned'),matches

def validate_payload(payload,path,expected_sha):
    """Compile from frozen source; never trust payload self-declarations."""
    require(hashlib.sha256(Path(path).read_bytes()).hexdigest()==expected_sha,'Source changed since review')
    _,expected,_=compile_model(path)
    require(payload==expected,'Payload differs from exact compiled source')
    require(payload['datasetId']==DS and payload['prefix']==PREFIX,'Wrong namespace')
    for n in payload['nodes']: require(set(n['labels'])<=ALLOWED_LABELS,'Unexpected label')
    for r in payload['relationships']: require(r['type'] in ALLOWED_RELS,'Unexpected relationship')
    return True

    q['Q06']='CALL { '+q['Q05']+' } RETURN collect({example:example,courtyard:courtyard,pattern:patternKey,status:resultStatus,candidateType:candidateType,ruleVersion:ruleVersion,cacheConsistent:cacheConsistent,boundaryIds:boundaryIds,boundaryKinds:boundaryKinds,boundaryStates:boundaryStates,participants:participants,sourceLocators:sourceLocators,geometryVersion:geometryVersion,inputModelVersion:inputModelVersion}) AS d06Comparison'
    q['Q07']='''MATCH (a)-[r]->(b) WHERE (a.caseId=$caseId OR b.caseId=$caseId) AND NOT (a.caseId=$caseId AND b.caseId=$caseId) RETURN count(r) AS crossDatasetEdges'''
    q['Q08']='''MATCH (n:D03V01) WITH labels(n) AS labels,count(*) AS n RETURN labels,n ORDER BY labels'''
    return q

"""First-create or verify D03V01 via configured MCP; never update other scopes.
Requires an explicit frozen model hash and mode. No blind retries or upserts.
"""
import argparse,asyncio,copy,datetime,hashlib,json
from pathlib import Path
from compile_model import compile_model,query_params,validate_payload,require,IMPORT,queries,DS,PREFIX,MODEL
from mcp_client import connected,snapshot,save,digest,canonical

def owned(row):
    p=row['properties']
    return p.get('caseId')==DS or p.get('datasetId')==DS or row['key'].startswith(PREFIX)

def split(g):
    new={k:[x for x in g[k] if owned(x)] for k in ['nodes','relationships']}
    old={k:[x for x in g[k] if not owned(x)] for k in ['nodes','relationships']}
    return canonical(new),canonical(old)

def expected_type(v):
    if isinstance(v,bool):return 'BOOLEAN NOT NULL'
    if isinstance(v,str):return 'STRING NOT NULL'
    if isinstance(v,int):return 'INTEGER NOT NULL'
    if isinstance(v,float):return 'FLOAT NOT NULL'
    if isinstance(v,list):
        return 'LIST<'+expected_type(v[0])+'> NOT NULL' if v else None
    raise ValueError(type(v))

def verify_exact(g,payload):
    for category in ['nodes','relationships']:
        expected={x['properties']['uid']:x for x in payload[category]}
        actual={x['key']:x for x in g[category]}
        require(len(actual)==len(g[category]) and set(actual)==set(expected),'Identity/count mismatch '+category)
        for uid,e in expected.items():
            a=actual[uid];require(a['properties']==e['properties'],'Property mismatch '+uid)
            if category=='nodes':require(sorted(a['labels'])==sorted(e['labels']),'Label mismatch '+uid)
            else:require(all(a[k]==e[k] for k in ['source','target','type']),'Endpoint/type mismatch '+uid)
            types={x['key']:x['type'] for x in a['types']}
            for k,v in e['properties'].items():
                t=expected_type(v)
                if t:require(types[k]==t,'Native type mismatch '+uid+'.'+k+': '+types[k]+' expected '+t)
    return {'exactNodes':len(g['nodes']),'exactRelationships':len(g['relationships']),'propertiesAndNativeTypes':'equal'}

async def main(args):
    d=Path(args.workspace).resolve(); out=d/'db'; path=Path(args.model) if args.model else d/'candidates'/MODEL
    model,payload,assignments=compile_model(path)
    validate_payload(payload,path,args.model_sha)
    receipt=out/'execution-stage.json'
    if receipt.exists():
        previous=json.loads(receipt.read_text())
        require(previous['modelSHA256']==args.model_sha,'Reconcile different execution source')
        require(args.mode=='verify','Prior execution exists; inspect/read back with verify, do not resend')
    async with connected(out) as call:
        before,summary=await snapshot(call,out/('write-before' if args.mode=='write' else 'verify-before'))
        new,old=split(before)
        if args.mode=='write':
            require(not new['nodes'] and not new['relationships'],'Namespace not empty; use verification, not replacement')
            baseline=json.loads((out/'post-guard/graph.json').read_text())
            require(before==baseline,'Database changed since rejection tests')
            schema=json.loads((out/'write-before/schema.json').read_text())
            require(any(c['labelsOrTypes']==['KGItem'] and c['properties']==['uid'] and c['type']=='UNIQUENESS' for c in schema['constraints']),'Required existing uniqueness constraint absent')
            require(all(i['state']=='ONLINE' for i in schema['indexes']),'Indexes not ONLINE')
            save(receipt,{'stage':'write-started-no-retry','modelSHA256':args.model_sha,'payloadSHA256':digest(payload),'compilerSHA256':hashlib.sha256(Path(__file__).with_name('compile_model.py').read_bytes()).hexdigest(),'beforeFingerprint':digest(before),'at':datetime.datetime.now().astimezone().isoformat()})
            validate_payload(payload,path,args.model_sha)
            result=await call(IMPORT,payload,write=True,receipt=out/'import-return.json')
            require(result==[{'nodesCreated':len(payload['nodes']),'relationshipsCreated':len(payload['relationships'])}],'Import rejected or unexpected returned counts; read back without retry')
        after,summary_after=await snapshot(call,out/'after')
        new,old_after=split(after)
        baseline=json.loads((out/'write-before/graph.json').read_text())
        require(old_after==baseline,'Existing graph fingerprint changed')
        require(json.loads((out/'after/schema.json').read_text())==json.loads((out/'write-before/schema.json').read_text()),'Constraints or indexes changed')
        exact=verify_exact(new,payload);save(out/'D03-readback.json',new)
        results={k:await call(q,query_params(payload),receipt=out/(k+'-raw.json')) for k,q in queries().items()}
        require(len(results['Q01'])==4 and len(results['Q02'])==16 and len(results['Q03'])==4,'Missing functional rows')
        require(all(x['valid'] for x in results['Q03']),'Entrance validity failed')
        require([x['relativePosition'] for x in results['Q03']]==[0.5,0.25,0.5,0.5],'Entrance results differ')
        require(len(results['Q04'])==26 and len(results['Q05'])==4,'Missing provenance/classification rows')
        require(all(x['cacheConsistent'] for x in results['Q05']),'Classification cache mismatch')
        require([(r['example'],r['resultStatus'],r['candidateType']) for r in results['Q05']]==[(x['example'],x['status'],x['candidateType']) for x in assignments],'Classification results differ')
        require(results['Q07']==[{'crossDatasetEdges':0}],'Cross dataset edge')
        save(out/'query-results.json',results)
        negative=[]
        for kind in ['wrong-model-hash','different-rule-content']:
            p=query_params(payload)
            if kind=='wrong-model-hash':p['modelSha']='not-the-source'
            else:p=copy.deepcopy(p);p['expectedRule']['definitionJson']='{}'
            r=await call(queries()['Q05'],p)
            require(len(r)==4 and all(x['resultStatus']=='invalid-input' for x in r),'Read-only mismatch not rejected')
            negative.append({'id':kind,'returned':r,'passed':True})
        save(out/'readback-negative-tests.json',negative)
        report={'status':'verified-technical-only-user-review-pending','at':datetime.datetime.now().astimezone().isoformat(),'modelSHA256':args.model_sha,**exact,'globalNodes':summary_after['nodes'],'globalRelationships':summary_after['relationships'],'oldNodes':len(baseline['nodes']),'oldRelationships':len(baseline['relationships']),'oldFingerprint':digest(baseline),'preservedFingerprint':digest(old_after),'oldGraphUnchanged':True,'constraintsIndexesUnchanged':True,'queries':{k:len(v) for k,v in results.items()},'readOnlyMismatchTests':len(negative),'sourceRuleBoundary':'four synthetic examples; not empirical validity; no native Neo4j GUI check'}
        save(out/'verification.json',report)
        stage=json.loads(receipt.read_text());stage.update(stage='verified',verificationFingerprint=digest(report));save(receipt,stage)
        print(json.dumps(report,ensure_ascii=False))

if __name__=='__main__':
    p=argparse.ArgumentParser(description=__doc__);p.add_argument('--workspace',required=True);p.add_argument('--model');p.add_argument('--model-sha',required=True);p.add_argument('--mode',choices=['write','verify'],required=True)
    asyncio.run(main(p.parse_args()))

import asyncio,copy,json
from pathlib import Path
from compile_model import CORE,RETURN_CLASS,IMPORT,queries,compile_model,MODEL,query_params,classify,validate_payload
from mcp_client import connected,snapshot,save,digest
D=Path(__file__).parent

async def main():
    model,payload,assignments=compile_model(D/'candidates'/MODEL)
    rule=next(n['properties'] for n in payload['nodes'] if n['id']=='R01')
    courts=[n['properties'] for n in payload['nodes'] if 'Courtyard' in n['labels']]
    prefix='UNWIND $courts AS c WITH c,$rule AS scheme,c.inputValid AS inputValid,c.cachedType AS cachedType,c.cachedStatus AS cachedStatus\n'
    q=prefix+CORE+RETURN_CLASS
    tested=[]
    async with connected(D/'db') as call:
        compilations={}
        compilations['import']=await call('EXPLAIN '+IMPORT,payload,write=True)
        for k,query in queries().items(): compilations[k]=await call('EXPLAIN '+query,query_params(payload),write=True)
        save(D/'db/compile-check.json',{'results':compilations,'meaning':'EXPLAIN returned without errors; no plan body supplied by MCP; not a performance review'})
        for c in courts:
            c=copy.deepcopy(c);c.update(inputValid=True,cachedType='',cachedStatus='')
            r=(await call(q,{'courts':[c],'rule':rule}))[0]
            expected=next(a for a in assignments if a['example']==c['exampleId'])
            assert r['resultStatus']==expected['status'] and r['candidateType']==expected['candidateType'],r
            tested.append({'id':'normal-'+c['exampleId'],'actual':r,'passed':True})
        base=copy.deepcopy(courts[0]);base.update(inputValid=True,cachedType='',cachedStatus='')
        cases=[]
        c=copy.deepcopy(base);c['boundaryKinds']=['building']*4;cases.append(('known-unmatched',c,rule,'known-unmatched'))
        c=copy.deepcopy(base);c['boundaryKinds']=['building','wall','building','open'];cases.append(('opposite-buildings-not-K02',c,rule,'known-unmatched'))
        c=copy.deepcopy(base);c['boundaryKinds']=c['boundaryKinds'][1:]+c['boundaryKinds'][:1];cases.append(('rotation',c,rule,'candidate-assigned'))
        c=copy.deepcopy(base);c['boundaryIds'].append('extra');c['boundaryKinds'].append('open');c['boundaryStates'].append('known');cases.append(('five-sides-outside-rule',c,rule,'out-of-scope'))
        c=copy.deepcopy(base);c['inputValid']=False;cases.append(('invalid-evidence-input',c,rule,'invalid-input'))
        c=copy.deepcopy(base);c['boundaryIds']=['S','E','N','W'];cases.append(('wrong-side-order',c,rule,'invalid-input'))
        r=copy.deepcopy(rule);r['patternKeys'].append('|'.join(base['boundaryKinds']));r['patternCodes'].append('CONFLICT-RULE');cases.append(('exclusive-rule-conflict',base,r,'rule-conflict'))
        for name,key,value in [
            ('short-kinds','boundaryKinds',base['boundaryKinds'][:3]),
            ('invalid-state-enum','boundaryStates',['known','known','known','unkown']),
            ('unknown-kind-known-state','boundaryKinds',['building','building','wall','unknown']),
            ('known-kind-unknown-state','boundaryStates',['known','known','known','unknown']),
            ('null-kinds','boundaryKinds',None),('null-states','boundaryStates',None),('null-ids','boundaryIds',None),
            ('string-kinds','boundaryKinds','building'),('null-array-element','boundaryKinds',['building','building','wall',None]),
            ('duplicate-ids','boundaryIds',['N','E','S','S']),('missing-shape','shapeCode',None),('empty-ids','boundaryIds',[])]:
            c=copy.deepcopy(base);c[key]=value;cases.append((name,c,rule,'invalid-input'))
        for name,c,rule2,expected in cases:
            result=(await call(q,{'courts':[c],'rule':rule2}))[0]
            assert result['resultStatus']==expected,(name,result)
            if expected!='candidate-assigned':assert result['candidateType']=='',result
            if name!='exclusive-rule-conflict':
                py=classify(c.get('boundaryKinds'),c.get('boundaryStates'),model['classificationScheme'],valid=c.get('inputValid'),shape=c.get('shapeCode'),ids=c.get('boundaryIds'))
                assert py[0]==expected,(name,py,expected)
            tested.append({'id':name,'actual':result,'passed':True})
        save(D/'db/classification-parameter-tests.json',{'kind':'actual read-only Cypher parameter tests; no persisted bad examples','tests':tested,'passed':len(tested)})
        before,_=await snapshot(call,D/'db/pre-guard')
        bad=[]
        for name in ['duplicate-node','duplicate-edge','missing-endpoint','wrong-namespace','missing-required-label','wrong-model-hash','param-namespace','extra-label','extra-reltype']:
            p=copy.deepcopy(payload)
            if name=='duplicate-node':p['nodes'][1]['properties']['uid']=p['nodes'][0]['properties']['uid']
            if name=='duplicate-edge':p['relationships'][1]['properties']['uid']=p['relationships'][0]['properties']['uid']
            if name=='missing-endpoint':p['relationships'][0]['target']='D03V01:DOES-NOT-EXIST'
            if name=='wrong-namespace':p['nodes'][0]['properties']['uid']='V01:NOT-A-WRITE'
            if name=='missing-required-label':p['nodes'][0]['labels'].remove('KGItem')
            if name=='wrong-model-hash':p['nodes'][0]['properties']['sourceModelSha256']='wrong'
            if name=='param-namespace':p['datasetId']='V01';p['prefix']='V01:'
            if name=='extra-label':p['nodes'][0]['labels'].append('Unapproved')
            if name=='extra-reltype':p['relationships'][0]['type']='UNAPPROVED';p['allowedRelationshipTypes'].append('UNAPPROVED')
            try: validate_payload(p,D/'candidates'/MODEL,payload['sourceModelSha256'])
            except ValueError: pass
            else: raise AssertionError('Local guard accepted '+name)
            returned=await call(IMPORT,p,write=True,receipt=D/'db'/('rejected-'+name+'.json'))
            assert returned==[],(name,returned)
            n=await call("MATCH (n) WHERE n.caseId='D03V01' OR n.uid STARTS WITH 'D03V01:' RETURN count(n) AS n")
            assert n==[{'n':0}],n
            bad.append({'id':name,'localGuardRejected':True,'returned':returned,'newNamespaceNodes':0,'passed':True})
        after,_=await snapshot(call,D/'db/post-guard')
        assert before==after
        save(D/'db/import-guard-tests.json',{'tests':bad,'fingerprintBefore':digest(before),'fingerprintAfter':digest(after),'unchanged':True})
    print(json.dumps({'EXPLAIN':len(compilations),'classificationParameterTests':len(tested),'rejectedPayloads':len(bad),'graphUnchanged':True},ensure_ascii=False))

if __name__=='__main__':asyncio.run(main())
