def dumps(v): return json.dumps(v,ensure_ascii=False,separators=(',',':'))
def patterns(scheme):
    out=[]
    for rule in scheme['rules']:
        p=rule['pattern']; variants=[p]
        if scheme['allowReflection']: variants.append(list(reversed(p)))
        keys=set()
        for v in variants:
            for i in range(len(v)) if scheme['allowRotation'] else [0]: keys.add('|'.join(v[i:]+v[:i]))
        out.extend((k,rule['code']) for k in sorted(keys))
    return out

def classify(kinds,states,scheme,valid=True,shape='rectangular-reference'):
    if not valid or len(kinds)!=len(states) or any(k not in KINDS for k in kinds): return 'invalid-input',[]
    if shape!=scheme['applicableShape'] or len(kinds)!=len(scheme['sideIds']): return 'out-of-scope',[]
    if any(s!='known' for s in states): return 'unknown',[]
    matches=sorted(set(code for key,code in patterns(scheme) if key=='|'.join(kinds)))
    return ('known-unmatched' if not matches else 'rule-conflict' if len(matches)>1 else 'candidate-assigned'),matches

def compile_model(path):

    sha=hashlib.sha256(raw).hexdigest(); nodes=[];rels=[]; scheme=model['classificationScheme']; pairs=patterns(scheme)
    def node(id,labels,p,example=None):
        common={'uid':PREFIX+id,'id':id,'caseId':DS,'datasetId':DS,'dataMode':'synthetic','recordStatus':'synthetic-assignment','sourceModelSha256':sha,'dataVersion':model['modelVersion']}
        if example: common['exampleId']=example
        props={**common,**p};nodes.append({'id':id,'labels':['KGItem','D03V01',*labels],'properties':props});return props
    def edge(id,type,source,target,props=None):
        p={'uid':PREFIX+id,'id':id,'caseId':DS,'datasetId':DS,'dataMode':'synthetic','sourceModelSha256':sha,'semanticOrigin':'research-extension',**(props or {})}
        rels.append({'id':id,'type':type,'source':PREFIX+source,'target':PREFIX+target,'properties':p});return p
    def source(id,p,pointer):
        excluded={'uid','id','caseId','datasetId','dataMode','recordStatus','sourceModelSha256','exampleId'}
        fields=sorted(set(p)-excluded)
        edge('SRC-'+id,'HAS_SOURCE',id,'EV01',{'fields':fields,'locator':pointer,'locatorKind':'JSON-pointer','sourceVersion':'1.0','evidenceRole':'synthetic-assignment','reviewStatus':'user-review-pending','projection':'字段由该JSON对象及其boundary数组投影；不是独立外部实证'})
    node('EV01',['Source','SyntheticSource'],model['source'])
    node('R01',['ClassificationScheme'],{'name':scheme['name'],'recordStatus':'research-candidate','ruleVersion':scheme['version'],'schemeMode':scheme['mode'],'applicableShape':scheme['applicableShape'],'sideIds':scheme['sideIds'],'patternKeys':[x[0] for x in pairs],'patternCodes':[x[1] for x in pairs],'typeCodes':[r['code'] for r in scheme['rules']],'typeNames':[r['name'] for r in scheme['rules']],'definitionJson':dumps(scheme),'sourceLocator':'/classificationScheme','reviewStatus':'user-review-pending'})

        assert len({b['id'] for b in bs})==len(bs)

        lengths=[math.dist(b['startXY'],b['endXY']) for b in bs]
        cp=node(c['id'],['Zone','Space','Courtyard'],{**c,'botClassIri':BOT+'Space','boundaryIds':[b['id'] for b in bs],'boundaryKinds':[b['kind'] for b in bs],'boundaryStates':[b['state'] for b in bs],'boundaryLengthsM':lengths,'boundarySegmentsJson':dumps(bs),'analysisItems':['D03-T01','D03-T02']},x);source(c['id'],cp,base)
        edge(x+'-CONTAINS-C','BOT_HAS_SPACE',u['id'],c['id'],{'botPropertyIri':BOT+'hasSpace','semanticOrigin':'BOT-aligned-synthetic-assertion','sourceUid':PREFIX+'EV01','sourceVersion':'1.0','locator':base,'fields':['source','target','type']})

            edge(x+'-CONTAINS-'+b['id'],'BOT_HAS_BUILDING',u['id'],b['id'],{'botPropertyIri':BOT+'hasBuilding','semanticOrigin':'BOT-aligned-synthetic-assertion','sourceUid':PREFIX+'EV01','sourceVersion':'1.0','locator':base,'fields':['source','target','type']})

## 1. 先把哪些东西分开

| 信息 | 本例落地 | 为何这样处理／不做什么 |
|---|---|---|
| 宅院整体与院落 | A-U01等整体Zone；A-C01等院落Space | 院落不硬挂北侧建筑；整体不等于其中一栋 |
| 参与围合的建筑与院墙 | 复用Building／Wall身份 | 建筑朝院一侧不冒称已识别实体墙；院墙不通过PART_OF挂在院落空间上 |
| 边界段 | 院落上的有序局部记录；N／E／S／W只在所属院落内标识 | 不建16个边段节点，不建尺寸节点；四侧只是本轮规则范围 |
| 哪个对象参与哪侧 | 院落→对象的BOUNDARY_PARTICIPANT边，携带局部编号及出处 | 不把二维参与关系升级为BOT邻接、真实接触或结构支承 |
| 入口 | 有独立ID的EntranceInterface；两条带角色INTERFACE_OF指向院侧和宿主墙 | 为跨项引用和局部定位而建点，不是因为它“有尺寸”。外侧空间未知，不补点、不生成通行 |
| 尺寸和位置 | 入口起止里程直接归入口；参照长度由同一边段投影 | 起点、方向、单位和几何版本保留；不储存另一份尺寸记录 |
| 来源 | 一个共享EV01；对象HAS_SOURCE限定字段，关系直接带来源UID／定位／版本 | 这是人工设定追溯，不是独立验证，不为全部建筑事实背书 |
| 候选分类 | 一份共享R01方案；每院落用CLASSIFIED_BY边保存派生候选及版本 | 不建“类型实体”和“D06建筑”；对象身份不随归类改变 |

**D04、D05共同接口已说明，但未实例化：** D04以后引用相同建筑、入口及局部范围；D05引用相同建筑／院落，并带成员和分析范围。不能据本例宣称两个模块已建库。立面、材料区域和组团是否独立建点仍按独立身份／引用需求逐项判断。

## 2. 四个合成例具体检验什么

按N→E→S→W顺序读。S侧从东向西累计里程；同一参照边长12m。这些数值只是示例设置，没有地域或测绘依据。

| 例 | N／E／S／W的构成 | 候选归类 | 入口中心里程比 | 本例的作用 |
|---|---|---|---:|---|
| A | 建筑／建筑／院墙／建筑 | ENC-3B-1W | 0.50 | 三侧建筑＋一侧院墙的正例 |
| B | 建筑／建筑／院墙／建筑 | ENC-3B-1W | 0.25 | 与A同构成型；入口位置作为辅助变体，不另增围合类型 |
| C | 建筑／建筑／院墙／明确开放 | ENC-2B-1W-1O | 0.50 | 相邻两侧建筑＋一侧院墙＋一侧开放的对照 |
| D | 建筑／建筑／院墙／未知 | 暂不能判定 | 0.50 | 已知的入口位置可计算，但西侧未知，不能当开放或强分型 |

入口比值按 `(s_a+s_b)/2/L` 计算。0.50只说明该参照下中心位置，不设“居中”的通用容差或建筑学评价。未判断净空、门扇、障碍和行人情境，所以四例均不输出可通行结论。

### 候选规则的作用范围

- 分析单位是一处院落，不是建筑栋数或宅院总体类型。
- 规则只检查人工四侧矩形参照的侧类和排列；绝对方位、旋转和镜像不作定义性特征。C型要求两建筑侧相邻；相对两侧建筑不自动归入此型。
- 院墙上存在入口仍属于粗粒度wall侧。此规则不是净实体边界长度比例，也不是完整物理闭合判断；需要扣洞／混合侧分析时另定粒度。
- 边界未知→unknown；已知但不匹配→known-unmatched；互斥规则同时满足→rule-conflict；五侧或非本形态范围→out-of-scope；输入／来源／版本／关系不一致→invalid-input。
- 不将候选名称直接映射成“三合院”“四合院”或地方类型，不声称两条规则穷尽院落形态。允许不同规则方案，修改后要重算而非仅修改显示名。

## 3. D06真正收到什么

收到院落ID、逐侧原始记录与关系、来源位置、规则版本、满足／未满足条件、候选结果及未定状态。Q05重算并对照保存的归类；Q06输出D06比较底表；入口连续量由Q03独立返回。

由这四个**合成设定**能检验的逻辑是：A和B可以同型而有入口位置差异；C和A有明确边界构成差异；D不能与C合并为“开放型”。这只是比较路径可运行，不是景宁建筑学发现，更不产生类型频率、文化成因或民族专属性。
